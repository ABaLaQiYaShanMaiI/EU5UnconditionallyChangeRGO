RGO_toggle_cost = {
	type = situation

	ai_tick = never
	automation_tick = never
	show_message = no

	potential = {
	}

	allow = {
	}

	select_trigger = {
		looking_for_a = situation
		target_flag = recipient
		name = "choose_situation"
		column = { data = name }
		visible = {
			situation:UnconditionallyChangeRGO = this
			situation_is_active = yes
		}
	}
	effect = {
		scope:actor ?= {
			if = {
				limit = {
					RGO_opt_cost = yes
				}
				set_variable = {
					name = rgo_opt_cost
					value = 0
				}
			}
			else = {
				set_variable = {
					name = rgo_opt_cost
					value = 1
				}
			}
		}
	}
}
