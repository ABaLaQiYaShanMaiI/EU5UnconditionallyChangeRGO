RGO_change_mining = {
	type = situation

	ai_tick = never
	automation_tick = never
	show_message = no
	show_message_to_target = no

	# 花费 = price:rgo_change（10 金）x 市场价系数；花费开关关闭或地点无市场 => 0（免费）
	# 每个倍率都带 desc（ScaleReason），供确认框解释收费理由
	price = price:rgo_change
	price_modifier = {
		if = {
			limit = { RGO_price_tier_1 = yes }
			add = { desc = "RGO_PRICE_TIER" value = 0.5 }
		}
		else_if = {
			limit = { RGO_price_tier_2 = yes }
			add = { desc = "RGO_PRICE_TIER" value = 1 }
		}
		else_if = {
			limit = { RGO_price_tier_3 = yes }
			add = { desc = "RGO_PRICE_TIER" value = 1.6 }
		}
		else_if = {
			limit = { RGO_price_tier_4 = yes }
			add = { desc = "RGO_PRICE_TIER" value = 2.4 }
		}
		else_if = {
			limit = { RGO_price_tier_5 = yes }
			add = { desc = "RGO_PRICE_TIER" value = 3.4 }
		}
		else_if = {
			limit = { RGO_price_tier_6 = yes }
			add = { desc = "RGO_PRICE_TIER" value = 5 }
		}
		else = {
			add = { desc = "RGO_PRICE_FREE" value = 0 }
		}
	}

	# 分类开关：关闭后本按钮不再出现
	potential = {
		scope:actor ?= {
			RGO_cfg_cat_mining_default = yes
		}
	}

	# 冷却开关：冷却未结束时按钮不可用
	allow = {
		scope:actor ?= {
			NOT = { has_country_modifier = rgo_change_cooldown }
		}
	}

	# 1) 选局势：GUI 已代入 recipient，正常流程会自动跳过这一步
	select_trigger = {
		looking_for_a = situation
		target_flag = recipient
		name = "choose_situation"
		column = { data = name }
		visible = {
			situation:UnconditionallyChangeRGO = this
			situation_is_active = yes
		}
	}

	# 2) 选原材料：运行时按货物自身字段动态筛选（见 00_rgo_config.txt 的 RGO_goods_in_*）
	select_trigger = {
		looking_for_a = goods
		target_flag = target_good
		name = "RGO_select_goods"
		column = { data = name }
		none_available_msg_key = "RGO_no_goods_available"
		cache_targets = yes
		cache_interaction_source_list = yes
		interaction_source_list = {
			every_goods = {
				limit = {
					RGO_goods_in_mining = yes
				}
				add_to_list = source
			}
		}
	}

	# 3) 选地块：仅本国地块；地理条件开关开启时还要求该货物能作为此地原材料
	select_trigger = {
		looking_for_a = location
		target_flag = target_location
		name = "RGO_select_location"
		map_mode = raw_material
		column = { data = name }
		column = { data = raw_material }
		none_available_msg_key = "RGO_no_locations_available"
		interaction_source_list = {
			scope:actor = {
				every_owned_location = {
					add_to_list = source
				}
			}
		}
		visible = {
			RGO_geo_allows_target = yes
			NOT = { raw_material ?= scope:target_good }
		}
	}
	effect = {
		RGO_apply_change = yes
	}
}

	}
}

RGO_change_produced = {
	type = situation

	ai_tick = never
	automation_tick = never
	show_message = no
	show_message_to_target = no

	# 花费 = price:rgo_change（10 金）x 市场价系数；花费开关关闭或地点无市场 => 0（免费）
	# 每个倍率都带 desc（ScaleReason），供确认框解释收费理由
	price = price:rgo_change
	price_modifier = {
		if = {
			limit = { RGO_price_tier_1 = yes }
			add = { desc = "RGO_PRICE_TIER" value = 0.5 }
		}
		else_if = {
			limit = { RGO_price_tier_2 = yes }
			add = { desc = "RGO_PRICE_TIER" value = 1 }
		}
		else_if = {
			limit = { RGO_price_tier_3 = yes }
			add = { desc = "RGO_PRICE_TIER" value = 1.6 }
		}
		else_if = {
			limit = { RGO_price_tier_4 = yes }
			add = { desc = "RGO_PRICE_TIER" value = 2.4 }
		}
		else_if = {
			limit = { RGO_price_tier_5 = yes }
			add = { desc = "RGO_PRICE_TIER" value = 3.4 }
		}
		else_if = {
			limit = { RGO_price_tier_6 = yes }
			add = { desc = "RGO_PRICE_TIER" value = 5 }
		}
		else = {
			add = { desc = "RGO_PRICE_FREE" value = 0 }
		}
	}

	# 分类开关：关闭后本按钮不再出现
	potential = {
		scope:actor ?= {
			RGO_cat_produced_enabled = yes
		}
	}

	# 冷却开关：冷却未结束时按钮不可用
	allow = {
		scope:actor ?= {
			NOT = { has_country_modifier = rgo_change_cooldown }
		}
	}

	# 1) 选局势：GUI 已代入 recipient，正常流程会自动跳过这一步
	select_trigger = {
		looking_for_a = situation
		target_flag = recipient
		name = "choose_situation"
		column = { data = name }
		visible = {
			situation:UnconditionallyChangeRGO = this
			situation_is_active = yes
		}
	}

	# 2) 选原材料：运行时按货物自身字段动态筛选（见 00_rgo_config.txt 的 RGO_goods_in_*）
	select_trigger = {
		looking_for_a = goods
		target_flag = target_good
		name = "RGO_select_goods"
		column = { data = name }
		none_available_msg_key = "RGO_no_goods_available"
		cache_targets = yes
		cache_interaction_source_list = yes
		interaction_source_list = {
			every_goods = {
				limit = {
					RGO_goods_in_produced = yes
				}
				add_to_list = source
			}
		}
	}

	# 3) 选地块：仅本国地块；地理条件开关开启时还要求该货物能作为此地原材料
	select_trigger = {
		looking_for_a = location
		target_flag = target_location
		name = "RGO_select_location"
		map_mode = raw_material
		column = { data = name }
		column = { data = raw_material }
		none_available_msg_key = "RGO_no_locations_available"
		interaction_source_list = {
			scope:actor = {
				every_owned_location = {
					add_to_list = source
				}
			}
		}
		visible = {
			RGO_geo_allows_target = yes
			NOT = { raw_material ?= scope:target_good }
		}
	}
	effect = {
		RGO_apply_change = yes
	}
}
