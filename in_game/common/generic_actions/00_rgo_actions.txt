RGO_change_mining = {
	type = situation

	ai_tick = never
	automation_tick = never
	show_message = no
	show_message_to_target = no

	# 花费 = price:rgo_change（10 金）x 市场价系数；花费开关关闭或地点无市场 => 0（免费）
	price = price:rgo_change
	price_modifier = {
		if = { limit = { RGO_price_tier_1 = yes } value = 0.5 }
		else_if = { limit = { RGO_price_tier_2 = yes } value = 1 }
		else_if = { limit = { RGO_price_tier_3 = yes } value = 1.6 }
		else_if = { limit = { RGO_price_tier_4 = yes } value = 2.4 }
		else_if = { limit = { RGO_price_tier_5 = yes } value = 3.4 }
		else_if = { limit = { RGO_price_tier_6 = yes } value = 5 }
		else = {
			value = 0
		}
	}

	# 分类开关：关闭后本按钮不再出现
	potential = {
		scope:actor ?= {
			RGO_cfg_cat_mining_default = yes
		}
	}

	# 冷却开关：冷却未结束时按钮不可用
	allow = {
		scope:actor ?= {
			NOT = { has_country_modifier = rgo_change_cooldown }
		}
	}

	# 1) 选局势：GUI 已代入 recipient，正常流程会自动跳过这一步
	select_trigger = {
		looking_for_a = situation
		target_flag = recipient
		name = "choose_situation"
		column = { data = name }
		visible = {
			situation:UnconditionallyChangeRGO = this
			situation_is_active = yes
		}
	}

	# 2) 选原材料：本分类 13 种
	select_trigger = {
		looking_for_a = goods
		target_flag = target_good
		name = "RGO_select_goods"
		column = { data = name }
		none_available_msg_key = "RGO_no_goods_available"
		cache_targets = yes
		cache_interaction_source_list = yes
		interaction_source_list = {
			every_goods = {
				limit = {
					OR = {
						this = goods:alum
						this = goods:coal
						this = goods:copper
						this = goods:gems
						this = goods:goods_gold
						this = goods:iron
						this = goods:lead
						this = goods:marble
						this = goods:mercury
						this = goods:saltpeter
						this = goods:silver
						this = goods:stone
						this = goods:tin
					}
				}
				add_to_list = source
			}
		}
	}

	# 3) 选地块：仅本国地块，且必须能产出所选原材料
	select_trigger = {
		looking_for_a = location
		target_flag = target_location
		name = "RGO_select_location"
		map_mode = raw_material
		column = { data = name }
		column = { data = raw_material }
		none_available_msg_key = "RGO_no_locations_available"
		interaction_source_list = {
			scope:actor = {
				every_owned_location = {
					add_to_list = source
				}
			}
		}
		visible = {
			is_goods_valid_for_location = scope:target_good
			NOT = { raw_material ?= scope:target_good }
		}
	}
	effect = {
		RGO_apply_change = yes
	}
}
