# ============================================================
# 原产物全修改 —— 开关与价格阶梯配置
# 本文件由 tools\generate_rgo_actions.ps1 自动生成，请勿手改
# 真值源：tools\rgo_common.ps1（默认值、价格阈值与系数、冷却月数）
#
# 判定优先级：玩家在局势面板点开关会写入国家变量（rgo_var_*）；
#             有变量则以变量为准，没有变量才用下面的文件级默认值。
# ============================================================

# --- 花费开关（yes = 按市场价收费 / no = 全部免费） ---
RGO_cfg_cost_default = { always = no }

# --- 冷却开关（yes = 每次修改挂 6 个月冷却 / no = 无冷却） ---
RGO_cfg_cooldown_default = { always = no }

# --- 地理条件开关（yes = 只列出原版允许该货物的地点 / no = 不限地理条件） ---
RGO_cfg_geo_default = { always = no }

# --- 七个分类的默认按钮开关 ---
RGO_cfg_cat_mining_default     = { always = yes }
RGO_cfg_cat_gathering_default  = { always = yes }
RGO_cfg_cat_crops_default      = { always = yes }
RGO_cfg_cat_food_default       = { always = yes }
RGO_cfg_cat_plantation_default = { always = yes }
RGO_cfg_cat_wild_default       = { always = yes }
RGO_cfg_cat_produced_default   = { always = no }

# ============================================================
# 运行时判定（root 均为国家，可在任何国家作用域里直接调用）
# ============================================================

RGO_cost_enabled = {
	OR = {
		AND = {
			has_variable = rgo_var_cost
			var:rgo_var_cost > 0
		}
		AND = {
			NOT = { has_variable = rgo_var_cost }
			RGO_cfg_cost_default = yes
		}
	}
}

RGO_cooldown_enabled = {
	OR = {
		AND = {
			has_variable = rgo_var_cooldown
			var:rgo_var_cooldown > 0
		}
		AND = {
			NOT = { has_variable = rgo_var_cooldown }
			RGO_cfg_cooldown_default = yes
		}
	}
}

RGO_cat_produced_enabled = {
	OR = {
		AND = {
			has_variable = rgo_var_produced
			var:rgo_var_produced > 0
		}
		AND = {
			NOT = { has_variable = rgo_var_produced }
			RGO_cfg_cat_produced_default = yes
		}
	}
}

RGO_geo_enabled = {
	OR = {
		AND = {
			has_variable = rgo_var_geo
			var:rgo_var_geo > 0
		}
		AND = {
			NOT = { has_variable = rgo_var_geo }
			RGO_cfg_geo_default = yes
		}
	}
}

# ============================================================
# 地理条件限制（被各分类动作的「选地点」步骤调用）
#   开关关闭（默认）：恒为 yes —— 本国任意地点都能改成任意原材料
#   开关开启：沿用原版 is_goods_valid_for_location，即货物的 location_potential
#             （气候 / 植被 / 地形 / 区域等静态地理条件，原版在开局缓存）
# ============================================================
RGO_geo_allows_target = {
	OR = {
		NOT = {
			scope:actor ?= {
				RGO_geo_enabled = yes
			}
		}
		is_goods_valid_for_location = scope:target_good
	}
}

# ============================================================
# 价格阶梯（被各分类动作的 price_modifier 调用）
# RGO_chargeable：花费开关开启 + 地块处于某个市场内，才谈得上收费
# 阈值(<) -> 系数：0.75 / 1.5 / 2.5 / 3.5 / 6（其余为最高档） -> 0.5 / 1 / 1.6 / 2.4 / 3.4 / 5
# ============================================================
RGO_chargeable = {
	scope:actor ?= {
		RGO_cost_enabled = yes
	}
	exists = scope:target_location.market
}

RGO_price_tier_1 = {
	RGO_chargeable = yes
	"scope:target_good.price_in_market(scope:target_location.market)" < 0.75
}

RGO_price_tier_2 = {
	RGO_chargeable = yes
	"scope:target_good.price_in_market(scope:target_location.market)" < 1.5
}

RGO_price_tier_3 = {
	RGO_chargeable = yes
	"scope:target_good.price_in_market(scope:target_location.market)" < 2.5
}

RGO_price_tier_4 = {
	RGO_chargeable = yes
	"scope:target_good.price_in_market(scope:target_location.market)" < 3.5
}

RGO_price_tier_5 = {
	RGO_chargeable = yes
	"scope:target_good.price_in_market(scope:target_location.market)" < 6
}

RGO_price_tier_6 = {
	RGO_chargeable = yes
}

# ============================================================
# 货物分类判定（root = goods 作用域；被各分类动作的「选原材料」步骤调用）
#   动态判定：只读货物自身字段，不写死货物 id ——
#     goods_category = raw_material / produced
#     goods_method   = mining / farming / hunting / gathering / forestry
#     is_food        = 该货物是否提供食物
#   因此其它模组或游戏更新新增的货物会自动进入对应分类。
#   例外：种植园只能按原版 01_plantation_goods.txt 的清单（生成时写入）；
#         制成品排除 $RGO_EXCLUDED_GOODS（奴隶）。
# ============================================================

RGO_goods_in_mining = {
	goods_category = raw_material
	goods_method = mining
}

RGO_goods_in_gathering = {
	goods_category = raw_material
	goods_method = gathering
}

RGO_goods_in_crops = {
	goods_category = raw_material
	goods_method = farming
	NOT = { is_food = yes }
	NOT = { RGO_goods_in_plantation = yes }
}

RGO_goods_in_food = {
	goods_category = raw_material
	goods_method = farming
	is_food = yes
}

RGO_goods_in_plantation = {
	goods_category = raw_material
	goods_method = farming
	OR = { this = goods:cotton this = goods:sugar this = goods:tobacco }
}

RGO_goods_in_wild = {
	goods_category = raw_material
	NOT = { goods_method = mining }
	NOT = { goods_method = gathering }
	NOT = { goods_method = farming }
}

RGO_goods_in_produced = {
	goods_category = produced
	NOT = { OR = { this = goods:slaves_goods } }
}
