# 缺少它时 price_database 会在 error.log 里报：
#   Missing modifier type for price. rgo_change_cost_modifier
# 加上之后该报错消失，同时国家的“成本修正”类加成也能作用于本价格。

	color=bad
	percent=yes
	game_data={
		category=country
	}
